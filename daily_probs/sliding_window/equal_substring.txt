class Solution {
public:
    int equalSubstring(string s, string t, int c) {
        int j=0,i=0;
        int n=s.length();
        int mx = 0,df=0;
        while(j<n){
            df += abs(s[j]-t[j]);
            while(df>c){
                df -= abs(s[i]-t[i]);   
                i++;
            }
            mx = max(mx,j-i+1);
            j++;
        }
        return mx;
    }
};
